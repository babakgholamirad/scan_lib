clean_result = function(text)
	if text == null then
		return ""
	end if
	replace_text = ["decompiling source...\nsearching unsecure values...\n","Unsafe check: string copy in ","Unsafe check: loop in array ","<b>","</b>",". Buffer overflow."]
	for t in replace_text
		text = text.replace(t,"")
	end for
	text.replace("\n\n\n\n","\n")
	text.replace("\n\n\n","\n")
	text.replace("\n\n","\n")
	return text
end function

parse_scan_result = function(text)
	exploits = []
	current_exploit = null
	print "------------------------------------------------------------------"
	print clean_result(text).split("\n")
	
	for line in clean_result(text).split("\n")
		line = line.trim() 
		if line == "" then
			continue
		else if line[:2] == "* " then
			if current_exploit then
				current_exploit.conditions.push(line)
			end if
		else
			if current_exploit then
				exploits.push(current_exploit)
			end if
			current_exploit = {"keyword": line, "conditions": []}
		end if
	end for
	if current_exploit then exploits.push(current_exploit)
	return exploits
end function

select_index = function(prompt, item_count)
	while true
		answer = user_input(prompt)
		if answer == null then return -1
		if answer.lower() == "q" then return -1
		index = item_count - 1
		while index >= 0
			if answer == index + "" then return index
			index = index - 1
		end while
		print("Invalid selection. Enter a listed number or q to quit.")
	end while
end function

handle_computer_result = function(computer)
	passwd_file = computer.File("/etc/passwd")
	if not passwd_file then
		print("The target has no /etc/passwd file.")
		return
	end if
	if not passwd_file.has_permission("r") then
		print("Unable to read /etc/passwd with the permissions returned by this exploit.")
		return
	end if
	if passwd_file.content == null or passwd_file.content == "" then
		print("The target password file is empty.")
		return
	end if

	local_computer = get_shell.host_computer
	home_path = "/home/" + active_user
	local_home = local_computer.File(home_path)
	if not local_home then
		print("Your local home folder could not be found: " + home_path)
		return
	end if

	local_path = home_path + "/receivedPaswd.txt"
	local_file = local_computer.File(local_path)
	if local_file then
		answer = user_input("Overwrite " + local_path + "? (y/n): ")
		if answer.lower() != "y" then
			print("Keeping the existing local file.")
			return
		end if
		local_file.delete
	end if

	local_computer.touch(home_path, "receivedPaswd.txt")
	local_file = local_computer.File(local_path)
	if not local_file then
		print("Could not create " + local_path)
		return
	end if
	local_file.set_content(passwd_file.content)
	print("Password file saved to " + local_path)

	while true
		print("1. Open decipher")
		print("2. Remove the saved file")
		print("0. Return")
		choice = user_input("Option: ")
		if choice == "0" then
			return
		else if choice == "1" then
			decipher = local_computer.File("/bin/decipher")
			if not decipher then
				print("Could not find /bin/decipher on your computer.")
			else
				get_shell.launch("/bin/decipher", local_path)
			end if
		else if choice == "2" then
			local_file.delete
			print("Removed " + local_path)
			return
		else
			print("Choose 0, 1, or 2.")
		end if
	end while
end function

	exit("<b>Usage: " + program_path.split("/")[-1] + " [ip_address] [port] (new_password)</b>\nScans the remote library, lets you choose a discovered exploit, and runs it.")

	exit("<b>Usage: " + program_path.split("/")[-1] + " [ip_address] [port] (new_password)</b>\nScans the remote library, lets you choose a discovered exploit, and runs it.")
end if

metaxploit = include_lib("/lib/metaxploit.so")
if not metaxploit then
	metaxploit = include_lib(current_path + "/tools/metaxploit.so")
end if
if not metaxploit then exit("<color=#ff0000>Error: Unable to find metaxploit.so.</color>")

address = params[0]
port = params[1].to_int
new_password = null
if params.len == 3 then
	if params[2] != "" then
		new_password = params[2]
	end if
end if

net_session = metaxploit.net_use(address, port)
if not net_session then exit("<color=#ff0000>Error: Unable to connect.</color>")

print("The number of devices connected to it as a gateway: " + net_session.get_num_conn_gateway())
print("The number of ports forwarded from the remote device: " + net_session.get_num_portforward())
print("The number of users registered on the remote device: " + net_session.get_num_users())
print("Are there any active users on the system? " + net_session.is_any_active_user())
if net_session.is_any_active_user() then
	print("Is root user active on the system? " + net_session.is_root_active_user())
end if

meta_lib = net_session.dump_lib
if not meta_lib then exit("<color=#ff0000>Error: Could not retrieve the remote library.</color>")
if meta_lib.is_patched() then
	print("The library patch date is:")
	print(meta_lib.is_patched(true))
else
	print("The library has never been patched.")
end if

memory_addresses = metaxploit.scan(meta_lib)
if not memory_addresses then exit("The library scan failed.")
if memory_addresses.len() == 0 then exit("No vulnerable memory addresses were found.")

exploit_array = []
for memory in memory_addresses
	scan_text = metaxploit.scan_address(meta_lib, memory)
	found_exploits = parse_scan_result(scan_text)
	if found_exploits.len() > 0 then
		exploit_array.push({"memory": memory, "exploits": found_exploits})
	end if
end for

if exploit_array.len() == 0 then exit("No usable exploit options were found in the scanned library.")

while true
	print("\nDiscovered vulnerable memory addresses:")
	index = 0
	for item in exploit_array
		print("(" + index + ") " + item.memory)
		index = index + 1
	end for

	memory_choice = select_index("Choose a memory address (q to quit): ", exploit_array.len())
	if memory_choice < 0 then exit("Scan cancelled.")
	selected_memory = exploit_array[memory_choice]

	print("\nExploit options for " + selected_memory.memory + ":")
	j = 0
	//print "------------------------------------------------------------------"
	//print selected_memory.exploits
	//print "------------------------------------------------------------------"
	for item in selected_memory.exploits
		print("  (" + j + ") " + item.keyword)
		for condition in item.conditions
			print("    " + condition)
		end for
		j = j + 1
	end for

	exploit_choice = select_index("Choose an exploit (q to quit): ", j+1)
	if exploit_choice < 0 then exit("Exploit cancelled.")
	selected_exploit = selected_memory.exploits[exploit_choice]

	print("Running " + selected_exploit.keyword + " at " + selected_memory.memory + "...")
	if new_password != null then
		result = meta_lib.overflow(selected_memory.memory, selected_exploit.keyword, new_password)
	else
		result = meta_lib.overflow(selected_memory.memory, selected_exploit.keyword)
	end if

	if result == null then
		print("Exploit returned null; no usable result was produced.")
		continue
	else
		result_type = typeof(result)
		if result_type == "shell" then
			print("Exploit returned a shell. Opening terminal.")
			result.start_terminal
		else if result_type == "computer" then
			print("Exploit returned a computer object.")
			handle_computer_result(result)
		else if result_type == "file" then
			print("Exploit returned a file: " + result.path)
			if result.content != null then
				print(result.content)
				continue
			end if
		else if result_type == "number" then
			if result == 1 then
				if new_password != null then
					print("Password change succeeded: " + new_password)
				else
					print("Exploit succeeded.")
				end if
			else if result == 0 then
				print("Exploit did not succeed.")
				continue
			else
				print("Exploit returned numeric result: " + result)
				continue
			end if
		else
			print("Exploit returned unsupported type: " + result_type)
			print(result)
			continue
		end if
	end if
	exit()
end while
