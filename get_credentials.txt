// Usage: get_credential <ip> <username>

if params.len != 2 then
	print("Usage: get_credential <ip> <username>")
	exit
end if

target_ip = params[0]
target_user = params[1]

if is_valid_ip(target_ip) == false then
	print("Invalid IP address: " + target_ip)
	exit
end if

whois_result = whois(target_ip)
if typeof(whois_result) != "string" then
	print("whois did not return readable text.")
	exit
end if

admin_email = ""
whois_lines = whois_result.split("\n")
for line in whois_lines
	if line.lower().indexOf("email address:") == 0 then
		admin_email = line[14:].trim()
	end if
end for

if admin_email == "" then
	print("Could not find an administrative email address in whois output.")
	exit
end if

mail_address = user_mail_address()
if mail_address == null or mail_address == "" then
	print("Could not determine the email address for this account.")
	exit
end if

mail_password = user_input("Email password for " + mail_address + ": ", true)
mail = mail_login(mail_address, mail_password)
if typeof(mail) != "MetaMail" then
	print("Email login failed: " + mail)
	exit
end if

wait(2)
existing_mail = mail.fetch()
if typeof(existing_mail) != "list" then
	print("Could not fetch the mailbox: " + existing_mail)
	exit
end if
print("Initial inbox fetch returned " + existing_mail.len + " item(s).")


seen_mail = []
for mail_id in existing_mail
	seen_mail.push(mail_id)
end for

message = "Hello, I'm " + target_user + ". I don't know what happens today that I can't enter into my workstation.\n"
message = message + "I get the ""wrong password"" error all the time. I'm sure that I'm entering my password correctly. What I can do?\n"
message = message + "Thank you."

wait(2)
send_result = mail.send(admin_email, "Login issues", message)
if send_result != 1 then
	print("Could not send the email: " + send_result)
	exit
end if

print("Message sent to " + admin_email + ". Waiting for a reply...")
poll_number = 0
while true
	wait(2)
	poll_number = poll_number + 1
	print("[poll " + poll_number + "] Fetching mailbox...")
	current_mail = mail.fetch()
	if typeof(current_mail) == "list" then
		print("[poll " + poll_number + "] Fetch returned " + current_mail.len + " item(s).")
		for mail_id in current_mail
			print("[poll " + poll_number + "] Mail item: " + str(mail_id))
			if seen_mail.indexOf(mail_id) == null then
				print("[poll " + poll_number + "] New mail item found; waiting before reading it.")
				wait(2)
				mail_body = mail.read(mail_id)
				print("[poll " + poll_number + "] Read returned type " + typeof(mail_body) + ": " + str(mail_body))
				seen_mail.push(mail_id)
				if typeof(mail_body) == "string" and mail_body.indexOf("These are the credentials of your computer:") != null then
					print("[poll " + poll_number + "] Reply contains the expected credentials heading.")
					mail_lines = mail_body.split("\n")
					for mail_line in mail_lines
						print("[poll " + poll_number + "] Reply line: " + mail_line)
						if mail_line.indexOf("Password:") == 0 then
							credential_password = mail_line[9:].trim()
							if credential_password != "" then
								print("Password for " + target_user + ": " + credential_password)
								exit
							end if
						end if
					end for
					print("[poll " + poll_number + "] Expected heading found, but no non-empty Password: line was parsed.")
				else
					print("[poll " + poll_number + "] New message did not contain the expected credentials heading. Read result: " + str(mail_body))
				end if
			else
				print("[poll " + poll_number + "] Item already seen.")
			end if
		end for
	else
		print("[poll " + poll_number + "] Fetch returned type " + typeof(current_mail) + ": " + str(current_mail))
	end if
end while
